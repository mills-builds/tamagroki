cmake_minimum_required(VERSION 3.16)
project(grokface_sim C CXX)
set(CMAKE_CXX_STANDARD 17)
if(DEFINED ENV{LVGL_DIR})
  set(LVGL_DIR "$ENV{LVGL_DIR}")
else()
  set(LVGL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../lvgl-sim")
endif()
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
add_subdirectory(${LVGL_DIR} lvgl)
add_executable(grokface_sim main.cc ../../main/tamagroki/grok_face.cc)
target_include_directories(grokface_sim PRIVATE ../../main/tamagroki)
target_link_libraries(grokface_sim lvgl m)
